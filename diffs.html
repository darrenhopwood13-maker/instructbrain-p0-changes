+/**
+ * The defect these cover: allocating a trade wrote confirmed_at, so routing an
+ * item counted as approving its wording, and the publish gate -- blocked on
+ * unconfirmed items -- was satisfied by a decision nobody had made.
+ */
+describe("what withdraws a person's confirmation", () => {
+  it("withdraws it when the report wording changes", () => {
+    expect(withdrawsConfirmation({ finding_text: "A new description." })).toBe(true);
+  });
+
+  it("withdraws it when the remedy is rewritten", () => {
+    expect(withdrawsConfirmation({ remedial_text: "Rake out and repoint." })).toBe(true);
+  });
+
+  it("withdraws it when the severity or grade is changed", () => {
+    expect(withdrawsConfirmation({ severity: "high" })).toBe(true);
+    expect(withdrawsConfirmation({ condition_grade: "C2" })).toBe(true);
+  });
+
+  it("withdraws it when the evidence is corrected", () => {
+    expect(withdrawsConfirmation({ capture_fields: { location: "Level 2" } })).toBe(true);
+  });
+
+  it("does NOT withdraw it for allocation, which is a routing decision", () => {
+    expect(withdrawsConfirmation({ assigned_trade: "Bricklayer" })).toBe(false);
+  });
+
+  it("does NOT withdraw it for a date, a status or the lifecycle", () => {
+    expect(withdrawsConfirmation({ due_date: "2026-10-20" })).toBe(false);
+    expect(withdrawsConfirmation({ status: "snag" })).toBe(false);
+    expect(withdrawsConfirmation({ due_date_overridden: true })).toBe(false);
+    expect(withdrawsConfirmation({ lifecycle_state: "open" })).toBe(false);
+  });
+
+  it("does not withdraw it for an empty patch", () => {
+    expect(withdrawsConfirmation({})).toBe(false);
+  });
+
+  it("counts a confirmation as content-free, so approving cannot clear itself", () => {
+    for (const field of MATERIAL_CONTENT_FIELDS) {
+      expect(field).not.toBe("confirmed_at");
+      expect(field).not.toBe("confirmed_by");
+      expect(field).not.toBe("assigned_trade");
+    }
+    expect(withdrawsConfirmation({ confirmed_at: "2026-10-11T00:00:00Z" })).toBe(false);
+  });
+
+  it("covers every content field the report actually prints", () => {
+    // A field that reaches the client's report but is missing from this list is
+    // a field a person can change after approving, silently.
+    for (const field of [
+      "finding_text",
+      "remedial_text",
+      "severity",
+      "condition_grade",
+      "regulatory_reference",
+    ]) {
+      expect(MATERIAL_CONTENT_FIELDS).toContain(field);

+/**
+ * Who may read a share link, and what they get.
+ *
+ * This exists because the rule was in the wrong place. The public endpoint
+ * checked the report's status on the manual-photographic branch only, so an
+ * ordinary report's link served live content whatever its status -- including
+ * after `reopenReport` set the report back to `in_review`, which is what
+ * happens the moment you reopen a published report to correct one line. Every
+ * link already in a client's inbox then served the working copy.
+ *
+ * The rule, stated once and testable:
+ *   nothing outside the organisation reads a report that is not published.
+ *
+ * A manual photographic report additionally has no review page to serve -- it
+ * is shared as the finished PDF only -- but that is a shape decision, not an
+ * access one, so it is kept separate on purpose.
+ */
+export type ShareAccessInput = {

+-- Share links pinned to an issued version.
+--
+-- Publishing already freezes the report into report_versions, but report_shares
+-- had no way to say WHICH version a link points at. So the public page read live
+-- tables, and a link could quietly change its contents after it was sent --
+-- including showing the working copy the moment a report was reopened.
+--
+-- Additive and backwards compatible. Existing rows keep version_id NULL and the
+-- reader falls back to the report's current version at read time. Nothing is
+-- backfilled: guessing a version for an old link would be worse than resolving
+-- it at read time.
+--
+-- NOT APPLIED YET. The code that writes and reads this column must not ship
+-- before this migration has run, or every share link will break on a missing
+-- column. Apply deliberately, then land the pinning code.
+
+alter table public.report_shares
+  add column if not exists version_id uuid references public.report_versions(id) on delete set null;
+
+comment on column public.report_shares.version_id is
+  'The issued report version this link serves. NULL on links minted before version pinning; those resolve to the report''s current version at read time.';
+
+create index if not exists report_shares_version_idx on public.report_shares(version_id);
